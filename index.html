// Test 1: Validate that the request was successful
pm.test("Status code is 200", function () {
    // Status 200 means the page was loaded successfully
    pm.response.to.have.status(200);
});


// Test 2: Validate that the page language is English
pm.test("Page language is English", function () {
    var responseText = pm.response.text();

    // Search for the lang attribute inside the <html> tag
    // Example matches: <html lang="en">  /  <html lang="en-US">
    var langMatch = responseText.match(/<html[^>]*\slang=["']([^"']+)["']/i);

    // Make sure that the lang attribute exists
    pm.expect(langMatch).to.not.be.null;

    // Extract and normalize the language value (e.g. "en" or "en-US")
    var pageLang = langMatch[1].toLowerCase();

    // Verify that the language starts with "en"
    pm.expect(pageLang).to.match(/^(en\b|en-)/);
});

בדיקה 2 — שפת העמוד

בודקת את מאפיין lang שנמצא בתגית <html>. מאפיין זה מציין את שפת העמוד. לדוגמה: 

<html lang="en">
אם הערך של lang מתחיל ב-en, ניתן להסיק שהעמוד מוגדר באנגלית.
4

הסבר מפורט על ה-Regular Expression

רג'קס 1 — מציאת מאפיין השפה בתגית HTML

/<html[^>]*\slang=["']([^"']+)["']/i

הרג'קס מחפש בתוך קוד ה-HTML של הדף את תגית <html> ומוציא ממנה את ערך השפה.

<html התחלת תגית html
[^>]* כל תו עד >
\s רווח לפני lang
lang= שם המאפיין
["'] פתיחת מרכאות
([^"']+) capture group — ערך השפה
["'] סגירת מרכאות
חלק ברג'קס מה הוא עושה למה צריך אותו?
<html מחפש את המחרוזת המדויקת <html מבטיח שאנחנו מסתכלים רק על תגית ה-html הראשית ולא על תגיות אחרות
[^>]* מתאים לכל תו שאינו >, אפס פעמים או יותר מאפשר לדלג על מאפיינים אחרים כמו dir="rtl" שעשויים להופיע לפני lang
\s מחפש תו רווח (space, tab וכו') בין מאפיינים ב-HTML תמיד יש רווח — <html dir="rtl" lang="en">
lang= מחפש את המחרוזת המדויקת lang= מזהה את שם המאפיין הספציפי שאנחנו מחפשים
["'] מתאים לגרש כפול " או לגרש בודד ' ב-HTML מותר לכתוב גם lang="en" וגם lang='en'
([^"']+) Capture Group — תופס את הערך בין המרכאות זה החלק שמוחזר ב-langMatch[1] — כאן יהיה en או en-US
/i דגל case-insensitive גורם לכך שגם LANG="EN" יתאים (הגנה מפני HTML לא סטנדרטי)

דוגמאות — מה יתאים ומה לא:

✓ יתאים (match)
<html lang="en"> <html lang="en-US"> <html dir="ltr" lang="en"> <html lang='en-GB'>
✗ לא יתאים (no match)
<html lang="he"> <html lang="ar"> <div lang="en"> <htmllang="en">
Capture Group — הסוגריים ( ) ברג'קס הם "לכידה". כשקוראים langMatch[1], מקבלים רק את מה שנלכד בתוך הסוגריים — כלומר ערך השפה בלבד, בלי שאר תגית ה-html.

רג'קס 2 — אימות שהשפה היא אנגלית

/^(en\b|en-)/
חלק ברג'קס מה הוא עושה למה צריך אותו?
^ עוגן תחילת מחרוזת מבטיח שהמחרוזת מתחילה ב-en ולא מכילה אותה רק בתוכה
( | ) OR — אחת מהאפשרויות מאפשר שתי צורות לגיטימיות של קוד שפה אנגלית
en\b en + word boundary מתאים ל-"en" בלבד — מונע התאמה ל-"enu" או "england"
en- en ואחריו מקף מתאים לכל קוד אזורי: en-US, en-GB, en-AU וכו'
✓ יעבור את הבדיקה
"en" → en\b מתאים "en-US" → en- מתאים "en-GB" → en- מתאים "en-AU" → en- מתאים
✗ לא יעבור את הבדיקה
"he" → מתחיל ב-he "ar" → מתחיל ב-ar "fr" → מתחיל ב-fr "english" → לא en\b ולא en-

עדכון חשוב — למה עברנו לאתר דמו?

במהלך ההרצה מול האתר האמיתי https://www.gov.il/he התקבלה חסימת אבטחה של האתר: 403 Forbidden ובהמשך מסך Sorry, you have been blocked. זו אינה שגיאת קוד בבדיקות, אלא מנגנון הגנה של האתר הממשלתי שמזהה וחוסם תעבורת אוטומציה.

החלטת התיקון: לא עוקפים את מנגנון האבטחה של אתר ממשלתי. במקום זאת יצרנו עמוד דמו מקומי בתוך הפרויקט: cypress/fixtures/gov-demo/search.html. העמוד מדמה Header עם רכיב חיפוש, תיבת חיפוש, תוצאות חיפוש ובאנר Cookies, כך שהבדיקות ממשיכות לתרגל Cypress, Page Object Model ו-Fixtures בצורה יציבה ובטוחה.

בנוסף תוקנו בעיות סביבתיות שעלו בדרך: הגדרת ignoreDeprecations ב-tsconfig.json עבור TypeScript 6, והוספת הפקודה cy.dismissCookieBanner() גם ל-commands.js כדי ש-Cypress יטען אותה בוודאות.

BasePage.ts — מחלקת אב

מחלקת בסיס שמכילה פעולות משותפות לכל הדפים. כל Page Object יורש ממנה עם extends. 

// מחלקת האב — כל Page Object יורש ממנה
export default class BasePage {

  // visit() — פותח את הדף בדפדפן
  // path — כתובת הדף; אם לא מעבירים, נפתח עמוד הדמו המקומי
  visit(path: string = 'cypress/fixtures/gov-demo/search.html'): void {
    cy.visit(path);          // מנווט לכתובת
    this.waitForPageLoad();   // ממתין שהדף יסיים להיטען
  }

  // waitForPageLoad() — מאמת שהדף נטען במלואו לפני שנמשיך
  // readyState === 'complete' פירושו: כל המשאבים (תמונות, סקריפטים) נטענו
  protected waitForPageLoad(): void {
    cy.document().its('readyState').should('eq', 'complete');
  }

  // getByTestId() — מוצא אלמנט לפי מאפיין data-testid
  // data-testid הוא מאפיין HTML שמוסיפים רק לצורך בדיקות — לא נראה למשתמש
  protected getByTestId(testId: string): Cypress.Chainable<JQuery<HTMLElement>> {
    return cy.get(`[data-testid="${testId}"]`);
  }

  // getByAriaLabel() — מוצא אלמנט לפי תווית נגישות aria-label
  // aria-label עוזר לקוראי מסך — ניתן להשתמש בו גם כ-Selector יציב
  protected getByAriaLabel(label: string): Cypress.Chainable<JQuery<HTMLElement>> {
    return cy.get(`[aria-label="${label}"]`);
  }
}

SearchPage.ts — Page Object לרכיב החיפוש

יורש מ-BasePage ומכיל את כל ה-Selectors והפעולות הקשורות לחיפוש.

// מייבאים את מחלקת האב — SearchPage תירש את כל המתודות שלה
import BasePage from './BasePage';

class SearchPage extends BasePage {

  // ─── Selectors ────────────────────────────────────────────────
  // כל Selector מוגדר פעם אחת כאן. אם ה-HTML ישתנה — מתקנים רק כאן.
  // שימוש ב-get (TypeScript) מחזיר את האלמנט בכל קריאה מחדש — לא נשמר בזיכרון

  // כפתור פתיחת תיבת החיפוש שנמצא ב-Header של האתר
  // מחפש לפי class, לפי aria-label שמכיל "חיפוש", או לפי class נוסף
  get searchToggleBtn(): Cypress.Chainable<JQuery<HTMLElement>> {
    return cy.get('.search-toggle, [aria-label*="חיפוש"], button.search-btn').first();
  }

  // שדה הקלט שבו המשתמש מקליד את מונח החיפוש
  // מחפש לפי סוג הקלט (search), לפי placeholder, או לפי class מקונן
  get searchInput(): Cypress.Chainable<JQuery<HTMLElement>> {
    return cy.get('input[type="search"], input[placeholder*="חיפוש"], .search-input input').first();
  }

  // כפתור השליחה (submit) של טופס החיפוש
  get searchSubmitBtn(): Cypress.Chainable<JQuery<HTMLElement>> {
    return cy.get('button[type="submit"], .search-submit').first();
  }

  // הצעות ה-Autocomplete שמופיעות בזמן הקלדה
  get searchSuggestions(): Cypress.Chainable<JQuery<HTMLElement>> {
    return cy.get('.autocomplete-suggestions, .search-dropdown, [class*="suggestion"]');
  }

  // המיכל שמכיל את תוצאות החיפוש לאחר שליחת הטופס
  get resultsContainer(): Cypress.Chainable<JQuery<HTMLElement>> {
    return cy.get('.search-results, [class*="search-result"], .results-list');
  }

  // הודעה שמוצגת כאשר אין תוצאות חיפוש
  get noResultsMessage(): Cypress.Chainable<JQuery<HTMLElement>> {
    return cy.get('.no-results, [class*="no-result"]');
  }

  // ─── Actions ──────────────────────────────────────────────────
  // מתודות שמבצעות פעולות — הבדיקות קוראות להן ולא עובדות ישירות עם HTML

  // מנווט לעמוד הדמו המקומי (במקום gov.il האמיתי שחסום)
  visitHomePage(): void {
    this.visit('cypress/fixtures/gov-demo/search.html');
  }

  // לוחץ על כפתור החיפוש ומאמת שתיבת החיפוש נפתחה ונראית
  openSearchBar(): void {
    this.searchToggleBtn.click();               // לוחץ על הכפתור
    this.searchInput.should('be.visible');     // מחכה שהשדה יופיע
  }

  // מוחק תוכן קיים בשדה ומקליד מונח חיפוש חדש
  typeSearchTerm(term: string): void {
    this.searchInput.clear().type(term);  // clear() — מנקה; type() — מקליד
  }

  // שולח את הטופס בלחיצת Enter ('{enter}' הוא קוד מיוחד של Cypress)
  submitSearchByEnter(): void {
    this.searchInput.type('{enter}');
  }

  // מנקה את שדה החיפוש — הערך חוזר לריק
  clearSearchInput(): void {
    this.searchInput.clear();
  }

  // מאחד את כל שלבי חיפוש מלא לפעולה אחת:
  // פתיחת השורה → הקלדה → שליחה
  searchFor(term: string): void {
    this.openSearchBar();           // שלב 1: פותח את תיבת החיפוש
    this.typeSearchTerm(term);      // שלב 2: מקליד את מונח החיפוש
    this.submitSearchByEnter();      // שלב 3: שולח עם Enter
  }
}

// מייצאים מופע אחד בלבד (Singleton Pattern)
// כך כל הבדיקות משתמשות באותו אובייקט — לא יוצרים אחד חדש בכל בדיקה
export default new SearchPage();
עיקרון חשוב: ה-Page Object לא מכיל assertions (כגון .should() על תוצאות). Assertions שייכות לבדיקות בלבד. ה-POM אחראי רק על איך לבצע פעולה, לא על מה לאמת.

Fixtures — הפרדת נתונים מקוד

קובץ JSON שמכיל את נתוני הבדיקה בלבד. הבדיקות טוענות אותו עם cy.fixture().

{
  // validSearch — נתוני בדיקה תקינה רגילה
  "validSearch": {
    "term": "passport",              // מונח החיפוש שיוקלד בשדה
    "expectedMinResults": 1,        // מספר תוצאות מינימלי שצפויות
    "expectedUrlSegment": "search"  // מה שצריך להיות ב-URL לאחר חיפוש
  },

  // anotherValidSearch — בדיקה תקינה נוספת עם מונח שונה
  "anotherValidSearch": {
    "term": "health",               // מונח חיפוש שני לגיוון הבדיקות
    "expectedMinResults": 1         // גם כאן מצפים לתוצאה אחת לפחות
  },

  // shortSearch — קלט קצר לבדיקת הצעות Autocomplete
  "shortSearch": {
    "term": "pa",                   // 2 תווים בלבד — אמורים להפעיל הצעות
    "description": "partial term to trigger autocomplete suggestions"
  },

  // specialCharSearch — Edge Case: תווים מיוחדים שלא אמורים לשבור את הדף
  "specialCharSearch": {
    "term": "!@#$%",                // קלט חריג — בודק גבולות של המערכת
    "description": "should not crash the page"
  }
}
למה Fixtures?
אם רוצים לשנות את מילת החיפוש — משנים רק בקובץ JSON, לא מחפשים בקוד. ניתן להריץ את אותן הבדיקות עם נתונים שונים (Data Driven Testing).

search.cy.ts — 5 בדיקות אוטומטיות

// מייבאים את ה-Page Object — הבדיקות לא יגעו ישירות ב-HTML
import searchPage from '../pages/SearchPage';

// מגדירים את מבנה הנתונים שנטען מקובץ ה-Fixture (JSON)
// TypeScript דורש שנגדיר את הצורה של האובייקט מראש
interface SearchData {
  validSearch:        { term: string; expectedMinResults: number; expectedUrlSegment: string };
  anotherValidSearch: { term: string; expectedMinResults: number };
  shortSearch:        { term: string; description: string };
  specialCharSearch:  { term: string; description: string };
}

// describe — מגדיר קבוצת בדיקות עם שם תיאורי
describe('Gov.il | Header Search Component', () => {

  // משתנה שיקבל את הנתונים מה-Fixture לאחר טעינתו
  let searchData: SearchData;

  // before() — רץ פעם אחת לפני כל הבדיקות בקבוצה
  // טוען את ה-JSON מ-cypress/fixtures/search/searchData.json
  before(() => {
    cy.fixture<SearchData>('search/searchData').then((data) => {
      searchData = data;  // שומרים את הנתונים בתוך המשתנה החיצוני
    });
  });

  // beforeEach() — רץ לפני כל בדיקה בנפרד
  // מבטיח שכל בדיקה מתחילה ממצב נקי: דף טרי + ללא כרזת עוגיות
  beforeEach(() => {
    searchPage.visitHomePage();    // פותח את עמוד הדמו
    cy.dismissCookieBanner();      // סוגר כרזת עוגיות אם קיימת
  });

  // ─── בדיקה 1 ─────────────────────────────────────────────────
  // בודקת שכפתור החיפוש מוצג למשתמש ב-Header
  it('should display the search toggle button in the header', () => {
    searchPage.searchToggleBtn.should('be.visible');
    // should('be.visible') — מאמת שהאלמנט נראה על המסך (לא מוסתר ב-CSS)
  });

  // ─── בדיקה 2 ─────────────────────────────────────────────────
  // בודקת שלאחר לחיצה על הכפתור, שדה החיפוש מופיע ומאופשר
  it('should open the search input when the search button is clicked', () => {
    searchPage.searchToggleBtn.click();          // לוחצים על כפתור החיפוש
    searchPage.searchInput
      .should('be.visible')   // מאמת שהשדה מוצג
      .and('be.enabled');    // מאמת שהשדה לא disabled
  });

  // ─── בדיקה 3 ─────────────────────────────────────────────────
  // בודקת שהמשתמש יכול להקליד בשדה ושהערך נשמר נכון

commands.ts — פקודות מותאמות אישית (Custom Commands)

Custom Command היא פקודה שאנחנו כותבים בעצמנו ומוסיפים לרשימת הפקודות של Cypress. לאחר ההוספה, ניתן לקרוא לה בכל בדיקה בדיוק כמו פקודת Cypress רגילה — עם cy. לפניה.

למה Custom Command?
הסגירה של כרזת העוגיות חוזרת ב-beforeEach של כל בדיקה. במקום להעתיק את הלוגיקה הזו בכל מקום — מגדירים אותה פעם אחת כ-Custom Command ומשתמשים בה בפשטות עם cy.dismissCookieBanner(). 
// Cypress.Commands.add() — כך מגדירים פקודה מותאמת אישית
// הפרמטר הראשון הוא שם הפקודה כמחרוזת
// הפרמטר השני הוא הפונקציה שתרוץ כשנקרא לפקודה
Cypress.Commands.add('dismissCookieBanner', () => {

  // cy.get('body') — מקבלים את כל גוף הדף
  // .then($body => ...) — מריצים קוד jQuery רגיל בתוך הפונקציה
  cy.get('body').then(($body) => {

    // $body.find() — מחפש בתוך גוף הדף אלמנטים שתואמים אחד מה-Selectors האלה:
    // .cookie-banner — class רגיל של כרזת עוגיות
    // #onetrust-accept-btn-handler — כפתור "קבל הכל" של ספריית OneTrust
    // [class*="cookie"] button — כפתור בתוך אלמנט שמכיל "cookie" ב-class שלו
    const banner = $body.find(
      '.cookie-banner, #onetrust-accept-btn-handler, [class*="cookie"] button'
    );

    // בודקים אם נמצא כרזה — אם הדף לא מציג כרזה, לא עושים כלום
    // banner.length > 0 אומר: "נמצאו אלמנטים שתואמים"
    if (banner.length > 0) {
      // cy.wrap() — עוטף אובייקט jQuery חזרה לאובייקט Cypress
      // { force: true } — לוחץ גם אם האלמנט מוסתר חלקית או לא אינטראקטיבי
      cy.wrap(banner.first()).click({ force: true });
    }
  });
});

מושג מרכזי: cy.intercept()

cy.intercept() — יירוט בקשות רשת
פקודה שמאפשרת ל-Cypress "לתפוס" בקשות HTTP (XHR / Fetch) לפני שהן יוצאות לשרת. ניתן לעשות שתי פעולות שונות:
מצב שם מה עושים מתי משתמשים
Spy (ריגול) Monitoring מיירטים ומסתכלים — הבקשה ממשיכה לשרת האמיתי כאשר רוצים לבדוק בקשות מול שרת אמיתי
Stub (דמוי) Mocking מיירטים ומחזירים תגובה מדומה — הבקשה לא יוצאת לשרת כאשר אין שרת, או כשרוצים לשלוט בתגובה

בשאלה זו משתמשים ב-Stub — מחזירים תגובה מדומה עם statusCode: 200 ו-body מוגדר מראש מה-fixture. כך הבדיקות עובדות ללא שרת אמיתי.

// תחביר בסיסי:
cy.intercept('GET', '/api/cities', {
  statusCode: 200,
  body: { cities: ['Tel Aviv', 'Jerusalem'] }
}).as('getCities');
//  ↑ method   ↑ URL pattern   ↑ תגובה מדומה (stub)   ↑ alias
כלל זהב: cy.intercept() חייב לרוץ לפני הפעולה שמפעילה את הבקשה!
אם ה-intercept נרשם אחרי שהבקשה כבר יצאה — Cypress לא יתפוס אותה.

FilterPage.ts — מחלקת Page Object

מחלקה חדשה FilterPage יורשת מ-BasePage — בדיוק כמו SearchPage. מכילה selectors ומתודות; אין אסרציות בפנים. 

import BasePage from './BasePage';

// טיפוס מותאם — השמות האפשריים לפילטרים
type FilterName = 'cities' | 'categories' | 'accessibilitytype';

class FilterPage extends BasePage {

  // ─── Selectors ──────────────────────────────────────────

  // כפתור פתיחת לוח הפילטרים (≡)
  get filterToggleBtn(): Cypress.Chainable<JQuery<HTMLElement>> {
    return cy.get('.filter-toggle');
  }

  // לוח הפילטרים — מכיל את שלוש תיבות הסימון
  get filterPanel(): Cypress.Chainable<JQuery<HTMLElement>> {
    return cy.get('.filter-panel');
  }

  // ─── Actions ────────────────────────────────────────────

  // פותח את עמוד הדמו המקומי
  visitFilterPage(): void {
    this.visit('cypress/fixtures/gov-demo/filters.html');
  }

  // לוחץ על ≡ ומחכה שהפאנל ייפתח
  openFilterPanel(): void {
    this.filterToggleBtn.click();
    this.filterPanel.should('be.visible');
  }

  // מסמן checkbox לפי שם הפילטר → מפעיל fetch() → Cypress מיירט
  selectFilter(name: FilterName): void {
    cy.get(`input[name="${name}"]`).check();
  }

  // מבטל סימון checkbox
  deselectFilter(name: FilterName): void {
    cy.get(`input[name="${name}"]`).uncheck();
  }
}

export default new FilterPage();  // Singleton — מופע יחיד

מושג מפתח: Cypress Command Queue

Cypress Command Queue (תור פקודות)
כל פקודות Cypress כמו cy.get(), cy.visit(), cy.click() לא מתבצעות מיידית. הן מתווספות לתור ביצוע ורצות רק כאשר Cypress מוכן — אחרי שהדף נטען, ורק כשהתור שלהן.

הבעיה בקוד: cy.get('.gLFyf') נרשם לתור הפקודות לפני cy.visit().
כשהתור מתבצע — Cypress מנסה למצוא את האלמנט על עמוד שעדיין לא נטען.

שלב 1: cy.get('.gLFyf') → נוסף לתור "חיפוש אלמנט"
SearchInputBox = Chainer (לא תוצאה מיידית!)

שלב 2: cy.visit(...) → נוסף לתור "טעינת עמוד"
שלב 3: .first().type(...) → נוסף לתור "הקלד"

── ביצוע התור ─────────────────────────────
✗ "חיפוש .gLFyf" → רץ לפני שהדף נטען! האלמנט לא קיים.
✓ "טעינת עמוד"
✓ "הקלד"
מצב קוד מה קורה
שגוי — get לפני visit const el = cy.get('.x');
cy.visit(url);
el.click();		 get רץ לפני טעינת הדף — האלמנט לא קיים
שרשרת תקינה cy.visit(url);
cy.get('.x').click();		 visit מסתיים לפני get — תקין
התשובה הנכונה: Malfunction in the order of operations — תקלה בסדר הפעולות.

הבעיה אינה בסלקטור, בערכים, או בתוכן הפקודות — אלא בכך ש-cy.get('.gLFyf') נרשם לתור הפקודות לפני cy.visit().
כשהתור רץ, Cypress מנסה למצוא את האלמנט על עמוד שעדיין לא נטען — ולכן נכשל.

הפתרון: תמיד לשים cy.visit() לפני cy.get().

שאלה 2 — מה יודפס בסוף ריצת הקוד?

1

הקוד שנתון בשאלה

int n1 = 55 , n2 = 66 , hint = 1 ;

for ( int i = 1; i <= n1; ++i)
{
   if (n1 % i == 0 && n2 % i == 0)
    hint = i;
}

לפני שמנתחים — צריך להבין שני מושגים: אופרטור Modulo ולולאת for.

2

אופרטור % (Modulo) — שארית חלוקה

a % b מחזיר את השארית של חלוקת a ב-b.

כאשר התוצאה היא 0 — b מחלק את a בדיוק, ללא שארית.

דוגמאות:
10 % 5 = 0  → 5 מחלק את 10 בדיוק
10 % 3 = 1  → 3 לא מחלק את 10 בדיוק (שארית 1)
55 % 11 = 0 → 11 מחלק את 55 בדיוק
66 % 11 = 0 → 11 מחלק את 66 בדיוק

תנאי ה-if בקוד בודק: "האם i מחלק בדיוק גם את n1 וגם את n2?" — כלומר, האם i הוא מחלק משותף.

3

מה הלולאה עושה בפועל?

הלולאה עוברת על כל ערך של i מ-1 עד 55. בכל איטרציה בודקת:
האם i מחלק את 55 וגם מחלק את 66 — אם כן, שומרת את i ב-hint.
בסוף הלולאה, hint מכיל את הערך האחרון שעמד בתנאי.

מעקב אחרי ערכים מעניינים בלולאה:

i 55 % i == 0 ? 66 % i == 0 ? נכנס ל-if ? מה קורה ל-hint?
1 כן כן כן hint = 1
5 כן לא לא נשאר 1
6 לא כן לא נשאר 1
11 כן כן כן hint = 11
55 כן לא לא נשאר 11
אופרטור && (AND): תנאי ה-if מתקיים רק כאשר שני התנאים נכונים. אם אחד מהם שגוי — לא נכנסים לגוף ה-if ו-hint לא משתנה.
4

למה התוצאה היא דווקא 11?

המחלקים המשותפים של 55 ו-66 הם: 1 ו-11.

רקע מתמטי — GCD (Greatest Common Divisor):
המחלק המשותף הגדול ביותר הוא המספר הגדול ביותר שמחלק בדיוק את שני המספרים.

55 = 5 × 11   |   66 = 6 × 11
המחלקים המשותפים: 1 ו-11. ה-GCD = 11.

הלולאה עוברת בסדר עולה (1 עד 55), ולכן בכל פעם שמוצאים מחלק משותף — hint מתעדכן לערך הגדול יותר. בסוף הלולאה, hint מכיל את ה-GCD — 11.

הפלט:  hint is: 11
התשובה הנכונה: □ 11

הלולאה מחשבת בפועל את ה-GCD (מחלק משותף גדול ביותר) של 55 ו-66. המחלקים המשותפים הם 1 ו-11. הלולאה עוברת בסדר עולה, כך ש-hint מתעדכן לבסוף ל-11.

2

שלוש דרכים לעבוד עם cy.intercept()

שיטה מה היא עושה מתי להשתמש
{ body: stub } מחליף את כל ה-body בנתונים קבועים שאלה 3 — בדיקת status code עם נתוני דמו
req.reply() stub מלא — Cypress עונה, לא פונה לשרת שאלה 4 — כשהשרת חסום (Cloudflare)
req.continue(res) מקבל תגובה אמיתית ואז משנה שדה ספציפי כשהשרת נגיש ומחזיר JSON תקין
למה לא משתמשים ב-req.continue() על gov.il?
gov.il מוגן ב-Cloudflare. בקשות אוטומטיות (Cypress, CI) מקבלות דף HTML של חסימה במקום JSON. req.continue() יחכה לתגובה אמיתית — ויקבל HTML, לא JSON — ואז res.body.title יכשל. לכן משתמשים ב-req.reply() עם דמו מקומי.
זרימה עם cy.intercept + req.reply(): דמו HTML → fetch('/api/.../prime_ministers_office') ↓ cy.intercept תופס את הבקשה ↓ req.reply({ body: { title: 'office' } }) ← לא פונה לשרת האמיתי ↓ הדמו מציג "office" ב-<h1> ↓ ✓ cy.wait('@getPMOffice').its('response.body.title').should('eq', 'office')
3

עמוד הדמו — prime-ministers.html

כמו בשאלות 2 ו-3, גם כאן נדרש דמו מקומי — קובץ HTML שמדמה את עמוד gov.il ושולח fetch() לאותו URL שרשמנו ב-cy.intercept. 

/* cypress/fixtures/gov-demo/prime-ministers.html */

<!-- כותרת המחלקה — מתמלאת מהAPI -->
<h1 data-testid="dept-title">טוען...</h1>

<script>
  // הדמו שולח fetch לאותו URL שמרשמים ב-cy.intercept
  fetch('https://www.gov.il/api/content/types/department/dcterms:identifier/prime_ministers_office')
    .then(function(res) { return res.json(); })
    .then(function(data) {
      // cy.intercept יירט את הבקשה ויחזיר { title: 'office' }
      // אז data.title יהיה 'office' — לא "משרד ראש הממשלה"
      document.querySelector('[data-testid="dept-title"]').textContent = data.title;
    });
</script>
הכלל: ה-URL בתוך fetch() בדמו חייב להתאים ל-URL pattern שרשמנו ב-cy.intercept. Cypress יורט כל בקשה שה-URL שלה מתאים ל-**/prime_ministers_office**.
4

intercept.cy.ts — קובץ הבדיקה

סדר הפעולות בבדיקה — חשוב: intercept חייב לרוץ לפני visit!

1. cy.intercept('GET', '**/prime_ministers_office**', req => req.reply({ body: { title: 'office' } })) → רישום המארב — Cypress ממתין לבקשה תואמת 2. cy.visit('cypress/fixtures/gov-demo/prime-ministers.html') → הדמו נפתח ושולח fetch() 3. cy.wait('@getPMOffice') → ממתין שה-intercept יתפוס את הבקשה 4. .its('response.body.title').should('eq', 'office') → ✓ מאמת שה-title בתגובה הוא 'office' 5. cy.get('[data-testid="dept-title"]').should('have.text', 'office') → ✓ מאמת שה-DOM התעדכן 
it('should change the page title via cy.intercept from "משרד ראש הממשלה" to "office"', () => {

  // 1. רישום intercept — req.reply() = stub מלא, לא פונה לשרת
  cy.intercept(
    'GET',
    '**/prime_ministers_office**',
    (req) => {
      req.reply({
        statusCode: 200,
        body: {
          title:       'office',   // ← השינוי שהשאלה מבקשת
          description: '',
          urlName:     'prime_ministers_office'
        }
      });
    }
  ).as('getPMOffice');

  // 2. פותח את הדמו המקומי
  cy.visit('cypress/fixtures/gov-demo/prime-ministers.html');

  // 3+4. ממתין לבקשה ומאמת שה-title הוא 'office'
  cy.wait('@getPMOffice')
    .its('response.body.title')
    .should('eq', 'office');

  // 5. מאמת שה-DOM עודכן
  cy.get('[data-testid="dept-title"]')
    .should('have.text', 'office');
});
